=== lib/app/(tabs)/layout.dart (new)
import 'package:fespalier/fespalier.dart';
import 'package:fespalier_adaptive/material.dart';
import 'package:flutter/material.dart';
import 'package:my_app/app.g.dart';

/// Which folder is which tab, left to right (without it the tabs follow folder order). Each
/// folder next to this file is one tab with a navigation stack and a state of its own.
const tabs = ['(home)', 'about'];

/// Asking for a `StatefulNavigationShell` instead of a `child` makes this a tab layout. The bar
/// is the menu: the `nav.dart` in each tab's folder says what the tab is called and its icon, and
/// `AdaptiveNavScaffold` shows `AppMenu.watch` as a `NavigationBar` on a phone, a
/// `NavigationRail` on a tablet and a `NavigationDrawer` on a wide window (docs/layouts.md,
/// "A bar, a rail or a drawer"). A tab keeps its state when the window changes size.
class TabsLayout extends StatelessWidget {
  const TabsLayout({super.key, required this.navigationShell});

  final StatefulNavigationShell navigationShell;

  @override
  Widget build(BuildContext context) => AdaptiveNavScaffold(
    shell: navigationShell,
    // `under:` the tab layout's folder, so each entry knows its tab: tapping the current one
    // goes back to its first page.
    menu: (ref) => AppMenu.watch(ref, under: '(tabs)'),
  );
}

=== test/adaptive_test.dart (new)
import 'package:fespalier/testing.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:my_app/app.g.dart';

/// Sizes the window in logical pixels (a test's own is 800 x 600).
void resize(WidgetTester tester, double width) {
  tester.view.physicalSize = Size(width, 900);
  tester.view.devicePixelRatio = 1;
  addTearDown(tester.view.reset);
}

void main() {
  testWidgets('the menu is a bar, a rail or a drawer by window width', (
    tester,
  ) async {
    resize(tester, 400);
    await pumpRouter(tester, AppRoutes.router());
    expect(find.byType(NavigationBar), findsOneWidget);
    expect(find.byType(NavigationRail), findsNothing);
    expect(find.byType(NavigationDrawer), findsNothing);

    // The destinations come from the nav.dart of each tab.
    await tester.tap(
      find.descendant(
        of: find.byType(NavigationBar),
        matching: find.text('About'),
      ),
    );
    await tester.pumpAndSettle();
    expect(currentLocation(tester), '/about');

    resize(tester, 800);
    await tester.pumpAndSettle();
    expect(find.byType(NavigationRail), findsOneWidget);
    expect(find.byType(NavigationBar), findsNothing);

    resize(tester, 1300);
    await tester.pumpAndSettle();
    expect(find.byType(NavigationDrawer), findsOneWidget);
    // Resizing keeps the tab.
    expect(currentLocation(tester), '/about');
  });
}
