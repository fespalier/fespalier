/// The OpenTelemetry SDK. A release build without an endpoint exports nothing: pass
/// --dart-define=OTEL_EXPORTER_OTLP_ENDPOINT=http://... to send spans to a collector.
final OtelZone observability = OtelZone(
  OtelZoneConfig(serviceName: serviceName, endpoint: FespalierOtel.endpoint()),
);

/// Brings the SDK up (it never throws), then runs [body]. This is the start of the process, not
/// of the app: a test that builds the app does not start the SDK, and the sink that startup()
/// installs reports once it is up.
Future<void> startObservability(Future<void> Function() body) async {
  WidgetsFlutterBinding.ensureInitialized();
  await observability.start(
    serviceVersion: '0.1.0',
    resourceAttributes: {...FespalierOtel.resourceAttributes},
  );
  await body();
}

/// Sentry starts first and runs the binding, startup() and runApp in its appRunner, so a
/// crash in any of them is Sentry's. An empty SENTRY_DSN turns Sentry off:
/// --dart-define=SENTRY_DSN=https://... sends events to your project.
Future<void> zone(Future<void> Function() body) => SentryFlutter.init(
  (options) => FespalierSentry.configure(
    options,
    dsn: const String.fromEnvironment('SENTRY_DSN'),
  ),
  appRunner: () => startObservability(body),
);

/// Runs once before the app (docs/app-startup.md); the providers it returns are
/// overridden in the app's ProviderScope, so the first frame already has them.
Future<List<Override>> startup() async {
  // One slot for every sink, before the router exists so the first navigation is reported.
  FespalierTelemetry.install(
    FespalierTelemetry.combine([
      FespalierOtel(isReady: () => observability.isReady),
      FespalierSentry(),
    ]),
  );
  return [
    // fespalier_auth: reads the stored session, with no network, so the first guard is synchronous.
    ...await restoreAuth(authSetup()),
    // fespalier_storage: a dataCache is saved here (null: nothing is saved).
    dataCacheStorage.overrideWithValue(await PrefsDataStorage.open()),
    // fespalier_connectivity: refetchOnReconnect follows the network.
    reconnectSignal.overrideWith(ConnectivitySignal.new),
    // fespalier_flags: where flag values come from (--dart-define=LABS=true turns /labs on).
    flagSource.overrideWithValue(
      const ConstFlags({'labs': bool.fromEnvironment('LABS')}),
    ),
    // fespalier_tolgee: the catalogs bundled in assets/i18n/ (a local read, never the network).
    translationsConfig.overrideWithValue(
      Translations(
        bundled: await BundledTranslations.load(locales: const ['en', 'fr']),
        baseLocale: 'en',
      ),
    ),
    // fespalier_image: which CDN the images come from (lib/images.dart).
    imageCdnProvider.overrideWithValue(appImages),
  ];
}

/// The providers startup.dart sets up, over a fake network (a widget test that reaches
/// connectivity_plus without one fails: docs/data.md, "Reconnects: fespalier_connectivity").
ProviderContainer start(FakeConnectivity network) {
  final container = ProviderContainer(
    overrides: [
      connectivitySource.overrideWithValue(network),
      reconnectSignal.overrideWith(ConnectivitySignal.new),
    ],
  );
  addTearDown(container.dispose);
  return container;
}

/// Ends a test: a container disposes what nothing listens to on a zero-duration timer, which
/// has to fire before testWidgets looks for timers left over.
Future<void> end(WidgetTester tester, ProviderContainer container) async {
  container.dispose();
  await tester.pump(const Duration(milliseconds: 1));
}

void main() {
  testWidgets('a network coming back fires reconnectSignal', (tester) async {
    final network = FakeConnectivity();
    final container = start(network);
    final fires = <int>[];
    container.listen(reconnectSignal, (_, count) => fires.add(count));
    await tester.pump();
    network.offline();
    expect(fires, isEmpty, reason: 'losing the network is not a reconnect');
    network.online();
    expect(fires, [1]);
    await end(tester, container);
  });

  testWidgets('hasNetwork is false only once the device says so', (
    tester,
  ) async {
    final network = FakeConnectivity();
    final container = start(network);
    container.listen(hasNetwork, (_, _) {});
    await tester.pump();
    expect(container.read(hasNetwork), isTrue);
    network.offline();
    expect(container.read(hasNetwork), isFalse);
    network.online();
    expect(container.read(hasNetwork), isTrue);
    await end(tester, container);
  });
}
