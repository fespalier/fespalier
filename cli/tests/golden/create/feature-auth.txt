import 'package:fespalier_auth/fespalier_auth.dart';

/// Where your identity provider plugs in. Every method throws until you write it: the sign-in
/// form shows the message, and nothing signs in. docs/auth.md has a backend over a JSON API, one
/// for OpenID Connect and Keycloak (`OidcBackend`), and the recipes for Firebase and Supabase.
class IdentityBackend extends AuthBackend {
  const IdentityBackend();

  static final UnimplementedError _todo = UnimplementedError(
    'connect your identity provider: docs/auth.md',
  );

  /// What a stored session is tied to: a session another backend stored is dropped at start.
  @override
  String get name => 'identity';

  @override
  Future<AuthSession> signIn(SignInRequest request) async => throw _todo;

  /// Throws `AuthRejected` when the server refused the refresh token (the session is over), and
  /// anything else when it could not ask (the session stays).
  @override
  Future<AuthSession> refresh(AuthSession session) async => throw _todo;

  @override
  Future<void> signOut(AuthSession session) async => throw _todo;
}

/// The app's [AuthConfig], read by `startup()` (restoreAuth) and by the tests.
///
/// `apiOrigins` is where the session is sent, and nowhere else: add your API's origin
/// (`Uri.parse('https://api.example.com')`) before calling it with `authHttpClient`.
AuthConfig authSetup() =>
    const AuthConfig(backend: IdentityBackend(), apiOrigins: <Uri>[]);

import 'package:fespalier/fespalier.dart';
import 'package:fespalier_auth/fespalier_auth.dart';

/// The input of the action, and so the fields of the sign-in form.
typedef SignInFields = ({String username, String password});

SignInFields form() => (username: '', password: '');

/// Checked on the device before the backend is asked.
FieldErrors? validate(SignInFields input) => FieldErrors({
  if (input.username.trim().isEmpty) 'username': 'Enter your user name',
  if (input.password.isEmpty) 'password': 'Enter your password',
});

/// A backend that throws `FieldErrors` for wrong credentials has them shown under their field.
Future<void> action(Ref ref, {required SignInFields input}) async {
  await ref
      .read(authSession.notifier)
      .signIn(
        PasswordSignIn(
          username: input.username.trim(),
          password: input.password,
        ),
      );
}

import 'package:fespalier/fespalier.dart';
import 'package:flutter/material.dart';
import 'package:my_app/app.g.dart';

/// Served at `/sign-in`. `SignInRoute.useForm` is generated from action.dart's form(), as on the
/// contact page of `--features forms`. The page does not navigate: the sign-in guard does.
class SignInPage extends HookConsumerWidget {
  const SignInPage({super.key, this.from});

  /// Where the guard that sent the user here was going.
  final String? from;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final form = SignInRoute.useForm(ref);
    final f = form.fields;
    return Scaffold(
      appBar: AppBar(title: const Text('Sign in')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          spacing: 16,
          children: [
            if (from != null) Text('Sign in to see $from'),
            TextField(
              controller: f.username.controller,
              decoration: InputDecoration(
                labelText: 'User name',
                errorText: f.username.error,
              ),
            ),
            TextField(
              controller: f.password.controller,
              obscureText: true,
              decoration: InputDecoration(
                labelText: 'Password',
                errorText: f.password.error,
              ),
            ),
            if (form.error case final e?) Text('$e'),
            FilledButton(
              // Null while the sign-in runs: the button is disabled.
              onPressed: form.onSubmit,
              child: Text(form.isPending ? 'Signing in...' : 'Sign in'),
            ),
          ],
        ),
      ),
    );
  }
}

=== test/auth_test.dart (new)
import 'package:fespalier/testing.dart';
import 'package:fespalier_auth/fespalier_auth.dart';
import 'package:fespalier_auth/testing.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:my_app/app.g.dart';
import 'package:my_app/auth_setup.dart';

void main() {
  testWidgets('signed out, /account goes to sign-in with where it was going', (
    tester,
  ) async {
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/account'),
      overrides: fakeAuth(),
    );
    expect(currentLocation(tester), '/sign-in?from=%2Faccount');
    expect(find.text('Sign in to see /account'), findsOneWidget);
  });

  testWidgets('signed in, /account opens, and signing out leaves it', (
    tester,
  ) async {
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/account'),
      overrides: fakeAuth(
        signedInAs: const AuthUser(id: 'ada-id', name: 'Ada'),
      ),
    );
    expect(currentLocation(tester), '/account');
    expect(find.text('Signed in as Ada'), findsOneWidget);

    await tester.tap(find.text('Sign out'));
    await tester.pumpAndSettle();
    expect(currentLocation(tester), '/sign-in?from=%2Faccount');
  });

  testWidgets('until you connect a provider, signing in says so', (
    tester,
  ) async {
    // The app's own backend, on an in-memory token store: nothing is stored, nothing is asked.
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/sign-in'),
      overrides: [
        authConfig.overrideWithValue(
          AuthConfig(backend: authSetup().backend, store: MemoryTokenStore()),
        ),
        authInitialState.overrideWithValue(const SignedOut()),
      ],
    );
    await tester.enterText(find.widgetWithText(TextField, 'User name'), 'ada');
    await tester.enterText(find.widgetWithText(TextField, 'Password'), 'pw');
    await tester.tap(find.widgetWithText(FilledButton, 'Sign in'));
    await tester.pumpAndSettle();
    expect(
      find.textContaining('connect your identity provider: docs/auth.md'),
      findsOneWidget,
    );
    expect(currentLocation(tester), '/sign-in');
  });
}
