=== lib/app/startup.dart (new)
import 'package:fespalier/startup.dart';
import 'package:fespalier_download/fespalier_download.dart';
import 'package:http/http.dart' as http;
import 'package:path_provider/path_provider.dart';

/// Where each base folder of a download is on this device. fespalier_download names no
/// plugin for it, so the app does (docs/downloads.md, "The foreground backend").
Future<String> appBases(DownloadBase base) async => switch (base) {
  DownloadBase.support => (await getApplicationSupportDirectory()).path,
  DownloadBase.cache => (await getApplicationCacheDirectory()).path,
  DownloadBase.documents => (await getApplicationDocumentsDirectory()).path,
};

/// Runs once before the app (docs/app-startup.md); the providers it returns are
/// overridden in the app's ProviderScope, so the first frame already has them.
List<Override> startup() => [
  // fespalier_download: the engine, over a foreground transfer (it stops with the app).
  downloadsEngine.overrideWithValue(
    Downloads(
      backend: HttpDownloadBackend(client: http.Client(), bases: appBases),
      store: FileDownloadStore(bases: appBases),
      files: TransferDownloadFiles(bases: appBases),
    ),
  ),
];

=== lib/downloads.dart (new)
import 'package:fespalier_download/fespalier_download.dart';

/// The file the sample page downloads: `--dart-define=DOWNLOAD_URL=https://files.myapp.dev/a.pdf`.
/// The default is a placeholder, so a run without it ends the download as failed.
const String downloadUrl = String.fromEnvironment(
  'DOWNLOAD_URL',
  defaultValue: 'https://files.example.com/sample.pdf',
);

/// A download is a request: an id, a URL and where the file goes, as a base folder and a relative
/// path (never an absolute one: an iOS container's path changes between launches). Add `bytes`
/// and `sha256` to have the file checked before it is moved into place. A URL that grants access
/// should be short-lived: never put a long-lived token in a header (docs/downloads.md, "Requests
/// and files").
final DownloadRequest sampleDownload = DownloadRequest(
  id: 'sample',
  url: Uri.parse(downloadUrl),
  file: const DownloadLocation(DownloadBase.support, 'downloads/sample.pdf'),
);

=== lib/app/downloads/page.dart (new)
import 'package:fespalier/fespalier.dart';
import 'package:fespalier_download/fespalier_download.dart';
import 'package:flutter/material.dart';
import 'package:my_app/downloads.dart';

/// Served at `/downloads`. Watching `downloads` opens the engine, which settles what a restart
/// left (lib/app/startup.dart builds it); `downloadStatus` rebuilds this only when this one
/// download changes. The engine here is the foreground one: a transfer stops when the app is
/// closed. For downloads that go on while it is closed, use fespalier_download_background
/// (docs/downloads.md, "The background backend"; it needs Flutter 3.47 and platform setup).
class DownloadsPage extends ConsumerWidget {
  const DownloadsPage({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    ref.watch(downloads);
    final status = ref.watch(downloadStatus(sampleDownload.id));
    final engine = ref.read(downloadsEngine);
    return Scaffold(
      appBar: AppBar(title: const Text('Downloads')),
      body: Center(
        child: switch (status) {
          Running(:final received, :final total) => Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              SizedBox(
                width: 200,
                child: LinearProgressIndicator(
                  value: total == null ? null : received / total,
                ),
              ),
              TextButton(
                onPressed: () => engine.pause(sampleDownload.id),
                child: const Text('Pause'),
              ),
            ],
          ),
          Paused() => FilledButton(
            onPressed: () => engine.resume(sampleDownload.id),
            child: const Text('Resume'),
          ),
          Queued() || Waiting() || Verifying() => const Text('Working...'),
          Complete() => const Text('Downloaded'),
          Failed() => FilledButton(
            onPressed: () => engine.retry(sampleDownload.id),
            child: const Text('Try again'),
          ),
          _ => FilledButton(
            onPressed: () => engine.start(sampleDownload),
            child: const Text('Download'),
          ),
        },
      ),
    );
  }
}

=== test/download_test.dart (new)
import 'package:fespalier/testing.dart';
import 'package:fespalier_download/fespalier_download.dart';
import 'package:fespalier_download/testing.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:my_app/app.g.dart';
import 'package:my_app/downloads.dart';

void main() {
  testWidgets('a download starts, shows its progress and ends', (tester) async {
    // No network and no disk: the fake backend plays the transfer.
    final backend = FakeDownloadBackend();
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/downloads'),
      overrides: downloadTestOverrides(backend: backend),
    );

    await tester.tap(find.text('Download'));
    await tester.pump();
    expect(backend.enqueued.single.id, sampleDownload.id);

    backend.emit(sampleDownload.id, const Running(50, 100));
    await tester.pump();
    expect(find.byType(LinearProgressIndicator), findsOneWidget);

    backend.emit(sampleDownload.id, Complete(sampleDownload.file, 100));
    await tester.pump();
    expect(find.text('Downloaded'), findsOneWidget);
  });

  testWidgets('a failed download can be tried again', (tester) async {
    final backend = FakeDownloadBackend();
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/downloads'),
      overrides: downloadTestOverrides(backend: backend),
    );
    await tester.tap(find.text('Download'));
    await tester.pump();

    backend.emit(sampleDownload.id, const Failed(DownloadFailure.network));
    await tester.pump();
    expect(find.text('Try again'), findsOneWidget);

    await tester.tap(find.text('Try again'));
    await tester.pump();
    expect(backend.enqueued, hasLength(2));
  });
}
