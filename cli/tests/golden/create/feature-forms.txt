import 'package:fespalier/fespalier.dart';

/// The input of the action, and so the fields of its form: a record with named fields.
typedef ContactFields = ({String name, String message});

/// The form starts empty. (A form that edits something takes the data it starts from.)
ContactFields form() => (name: '', message: '');

/// Checked on the device before the action runs, and live in the form after a first submit.
FieldErrors? validate(ContactFields input) => FieldErrors({
  if (input.name.trim().isEmpty) 'name': 'Enter your name',
  if (input.message.trim().isEmpty) 'message': 'Write a message',
});

/// The write. Send `input` to your server here; it can throw
/// `FieldErrors({'name': '...'})` to show an error under a field.
Future<void> action(Ref ref, {required ContactFields input}) async {}

=== lib/app/contact/page.dart (new)
import 'package:fespalier/fespalier.dart';
import 'package:flutter/material.dart';
import 'package:my_app/app.g.dart';

/// Served at `/contact`. `ContactRoute.useForm` is generated from action.dart's form(): a typed
/// field per field of the action's input, an error per field, and a submit that is null while the
/// action runs. It is a real hook, so the page is a [HookConsumerWidget].
class ContactPage extends HookConsumerWidget {
  const ContactPage({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final form = ContactRoute.useForm(ref);
    final f = form.fields;
    return Scaffold(
      appBar: AppBar(title: const Text('Contact')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          spacing: 16,
          children: [
            TextField(
              controller: f.name.controller,
              decoration: InputDecoration(
                labelText: 'Name',
                errorText: f.name.error,
              ),
            ),
            TextField(
              controller: f.message.controller,
              decoration: InputDecoration(
                labelText: 'Message',
                errorText: f.message.error,
              ),
            ),
            if (form.error case final e?) Text('$e'),
            FilledButton(
              // Null while the action runs: the button is disabled.
              onPressed: form.onSubmit,
              child: Text(form.isPending ? 'Sending...' : 'Send'),
            ),
          ],
        ),
      ),
    );
  }
}

=== test/forms_test.dart (new)
import 'package:fespalier/testing.dart';
import 'package:fespalier_forms/testing.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:my_app/app.g.dart';
import 'package:my_app/app/contact/action.dart';

void main() {
  test('an empty contact is refused with an error per field', () {
    expect(
      validate(form()),
      isFieldErrors({'name': 'Enter your name', 'message': 'Write a message'}),
    );
    expect(validate((name: 'Ann', message: 'Hi')), isFieldErrors({}));
  });

  testWidgets('submitting the empty form shows the errors under the fields', (
    tester,
  ) async {
    await pumpRouter(tester, AppRoutes.router(initialLocation: '/contact'));
    expect(find.text('Enter your name'), findsNothing);

    await tester.tap(find.text('Send'));
    await tester.pump();
    expect(find.text('Enter your name'), findsOneWidget);
    expect(find.text('Write a message'), findsOneWidget);

    await tester.enterText(find.widgetWithText(TextField, 'Name'), 'Ann');
    await tester.pump();
    expect(find.text('Enter your name'), findsNothing);
  });

  testWidgets('a valid form runs the action and shows no error', (
    tester,
  ) async {
    await pumpRouter(tester, AppRoutes.router(initialLocation: '/contact'));
    await tester.enterText(find.widgetWithText(TextField, 'Name'), 'Ann');
    await tester.enterText(find.widgetWithText(TextField, 'Message'), 'Hi');
    await tester.tap(find.text('Send'));
    await tester.pumpAndSettle();
    expect(find.text('Enter your name'), findsNothing);
    expect(find.text('Write a message'), findsNothing);
  });
}
