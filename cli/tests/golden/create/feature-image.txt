import 'package:fespalier_image/fespalier_image.dart';

/// Where the images are served from: `--dart-define=IMAGES_URL=https://img.example.com`. A public
/// address, never a key (docs/responsive-images.md, "Signed image URLs").
const String imagesUrl = String.fromEnvironment(
  'IMAGES_URL',
  defaultValue: 'https://images.example.com',
);

/// How the app asks for an image: a URL template, with the width the box needs in `{width}`.
/// Swap the builder for your CDN's (`ImgproxyUrlBuilder`, `CloudinaryUrlBuilder`, `ImgixUrlBuilder`,
/// `ThumborUrlBuilder`): docs/responsive-images.md, "URL builders".
const ImageCdn appImages = ImageCdn(
  builder: TemplateUrlBuilder('$imagesUrl/{source}?w={width}&q={quality}'),
  quality: 80,
);

=== lib/app/photo/page.dart (new)
import 'package:fespalier_image/fespalier_image.dart';
import 'package:flutter/material.dart';

/// Served at `/photo`. A `ResponsiveImage` measures its box, multiplies by the device pixel ratio
/// and asks the CDN (lib/images.dart) for the smallest width that is enough, so this square
/// downloads a few hundred pixels, not the original. A failed load shows the error view, whose
/// `retry` loads it again.
class PhotoPage extends StatelessWidget {
  const PhotoPage({super.key});

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('Photo')),
    body: Center(
      child: ResponsiveImage(
        'sample.jpg',
        width: 160,
        aspectRatio: 1,
        errorBuilder: (context, error, retry) => Center(
          child: TextButton(onPressed: retry, child: const Text('Retry')),
        ),
      ),
    ),
  );
}

=== test/image_test.dart (new)
import 'dart:ui' as ui;

import 'package:fespalier/testing.dart';
import 'package:fespalier_image/fespalier_image.dart';
import 'package:fespalier_image/testing.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:my_app/app.g.dart';
import 'package:my_app/images.dart';

void main() {
  late ui.Image image;
  setUpAll(() async => image = await createTestImage(width: 8, height: 8));

  testWidgets('the photo asks the CDN for the width its box needs', (
    tester,
  ) async {
    // Loads go through FakeImages: no network, and the test sees every URL.
    final images = FakeImages(image: image);
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/photo'),
      overrides: [imageCdnProvider.overrideWithValue(images.cdn(appImages))],
    );
    // A 160-point box at the test view's 3x pixel ratio is 480 pixels: the 640 bucket.
    expect(images.requested, ['$imagesUrl/sample.jpg?w=640&q=80']);
    expect(find.byType(RawImage), findsOneWidget);
  });

  testWidgets('a failed load shows the retry, and retrying asks again', (
    tester,
  ) async {
    final images = FakeImages();
    await pumpRouter(
      tester,
      AppRoutes.router(initialLocation: '/photo'),
      overrides: [imageCdnProvider.overrideWithValue(images.cdn(appImages))],
    );
    final url = images.requested.single;
    images.fail(url);
    await tester.pumpAndSettle(); // the error reaches the widget a frame later
    expect(find.text('Retry'), findsOneWidget);

    await tester.tap(find.text('Retry'));
    await tester.pump();
    expect(images.requested, [url, url]);
  });
}
