=== lib/app/app.dart (new)
import 'package:fespalier/fespalier.dart';
import 'package:flutter/material.dart';

/// The app around the router: theme, title, locales. The generated main() builds it
/// (lib/app.main.g.dart) and gives it the router.
class App extends StatelessWidget {
  const App({super.key, required this.router});

  final GoRouter router;

  @override
  Widget build(BuildContext context) =>
      MaterialApp.router(title: 'my_app', routerConfig: router);
}

=== lib/app/layout.dart (new)
import 'package:flutter/material.dart';

/// Wraps every page below this folder; `child` is the page.
///
/// The transparent [Material] gives ink widgets on a page (`ListTile`, `InkWell`)
/// something to paint on under the Scaffold's body.
class AppLayout extends StatelessWidget {
  const AppLayout({super.key, required this.child});

  final Widget child;

  @override
  Widget build(BuildContext context) => Scaffold(
    body: SafeArea(
      child: Material(type: MaterialType.transparency, child: child),
    ),
  );
}
